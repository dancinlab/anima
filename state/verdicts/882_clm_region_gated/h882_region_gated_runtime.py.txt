"""H_882 - region-gated plasticity (영역별 학습 게이트): gate edge-learn BY REGION.

H_861 found readout-only edge-learn forgets catastrophically (RETAIN z_drop 1.984
>= 1.0 FAIL) because the readout is SHARED across the base byte band and the new-
context high band; re-fitting it for the new band STEALS probability mass from the
base band. H_865 fixed BOUND with a trunk-adjacent additive adapter; H_872 mapped
the freeze-depth boundary. H_882 asks a DIFFERENT lever: GATE PLASTICITY BY REGION.

Turn the trainable edge ON only over the byte-region the new context actually
occupies (R_new), and OFF (FROZEN-readout pass-through) over the base-ability
byte-region (R_base). Hypothesis: region-gating reduces interference/forgetting
vs an UNGATED edge-learn of the SAME capacity, while keeping adaptation gain.

Two arms, IDENTICAL everything except the region gate (E5 variable-ablation):
  UNGATED : logits = readout(h + adapter(h))                  (adapter affects ALL 256 logits)
  GATED   : logits[R_new ] = readout(h + adapter(h))[R_new ]  (adapter ON  in R_new)
            logits[R_base] = readout(h)[R_base]               (adapter OFF in R_base, frozen pass-through)
  Both arms use the H_865 zero-init additive adapter (a real lever). The gate
  masks the adapter's OUTPUT-LOGIT contribution to R_new only -> R_base byte
  probability mass is protected BY CONSTRUCTION; backprop flows only through R_new.

Region R_new is DERIVED ONCE from the new-context TRAIN target bytes (>=99% mass
coverage), frozen for the run -> no test leakage.

Falsifier (FROZEN VERBATIM · prereg commit 54a2bae7d · post-tuning = 0)
  F-CLM-REGION-GATED-RETAIN : z_drop_gated <  z_drop_ungated     (gating reduces forgetting)
  F-CLM-REGION-GATED-GAIN   : gain_gated   >= gain_ungated        (gating keeps adaptation)
  🟢 SUPPORTED-NUMERICAL iff BOTH PASS; else 🔴 CLOSED-NEGATIVE (a_paper_negative_ok).

SW-sim of the non-deterministic on-chip edge-learn is acceptable at the measurement
rung (a_scale_honest_scope); foundation H_679 established HW edge-learn is real.
Scope: MEASUREMENT rung (mid d512/L8/E8) only.

NOTE: project hexa-native source policy stores this scaffold as .hexa; its body
is the executable harness (torch / CLM model import) run at the measurement rung.
"""

# --- frozen experiment constants (mirror H_865 / bf98c01 verbatim) ----------
BASE_SEED = 101
NEW_SEED = 202
N_ADAPT = 300
LR_EDGE = 3e-3
SEQ_LEN = 64
BATCH_SIZE = 16
N_EVAL_BATCHES = 32
ADAPTER_RANK = 64
REGION_MASS = 0.99            # R_new covers >= this fraction of new-context target mass
RUN_SEED = 882                # registered headline seed
SEED_SWEEP = [882, 42, 123, 7, 2026]


def _lcg(seed):
    state = seed & 0x7FFFFFFF
    while True:
        state = (1103515245 * state + 12345) & 0x7FFFFFFF
        yield state


def make_lane_bytes(kind, n, seed):
    rng = _lcg(seed + 1)
    out = []
    if kind == "web":
        motifs = [
            [0x68, 0x65, 0x6c, 0x6c, 0x6f, 0x20],
            [0x77, 0x6f, 0x72, 0x6c, 0x64, 0x2e, 0x20],
            [0x74, 0x68, 0x65, 0x20, 0x71, 0x75, 0x69],
        ]
        while len(out) < n:
            m = motifs[next(rng) % len(motifs)]
            out.extend(m)
            if next(rng) % 17 == 0:
                out.append(0x20 + (next(rng) % 0x5e))
    else:
        period = 11 + (seed % 7)
        base = 0x80
        for i in range(n):
            v = base + ((i % period) * 13 + (next(rng) % 5)) % 0x7f
            out.append(v & 0xFF)
    return out[:n]


def batchify(byte_list, seq_len, n_batches, batch_size, seed):
    g = torch.Generator().manual_seed(seed)
    data = torch.tensor(byte_list, dtype=torch.long)
    max_start = len(data) - seq_len - 1
    batches = []
    for _ in range(n_batches):
        starts = torch.randint(0, max_start, (batch_size,), generator=g)
        toks = torch.stack([data[s:s + seq_len] for s in starts])
        tgts = torch.stack([data[s + 1:s + seq_len + 1] for s in starts])
        batches.append((toks, tgts))
    return batches


def derive_region(new_train, vocab, mass):
    """R_new = smallest byte set covering >= `mass` of new-context target bytes.
    Frozen ONCE from the TRAIN slice (no test leakage)."""
    counts = torch.zeros(vocab, dtype=torch.float64)
    for _, tgts in new_train:
        counts += torch.bincount(tgts.reshape(-1), minlength=vocab).double()
    total = counts.sum().clamp(min=1.0)
    order = torch.argsort(counts, descending=True)
    cum = torch.cumsum(counts[order], dim=0) / total
    k = int((cum < mass).sum().item()) + 1
    r_new = order[:k]
    gate = torch.zeros(vocab, dtype=torch.bool)
    gate[r_new] = True
    return gate  # True = R_new (adapter ON), False = R_base (adapter OFF)


class AdapterEdge(nn.Module):
    def __init__(self, d_model, rank):
        super().__init__()
        self.down = nn.Conv1d(d_model, rank, kernel_size=1)
        self.act = nn.GELU()
        self.up = nn.Conv1d(rank, d_model, kernel_size=1)
        nn.init.zeros_(self.up.weight)
        nn.init.zeros_(self.up.bias)

    def forward(self, h):
        return h + self.up(self.act(self.down(h)))


class RegionGatedCLM(nn.Module):
    """H_865 adapter edge + optional OUTPUT-LOGIT region gate.

    gated=False -> ungated arm (adapter affects every byte logit, == H_865).
    gated=True  -> the adapter's logit contribution is restricted to R_new;
                   R_base logits are the FROZEN-readout values (adapter OFF).
    """

    def __init__(self, base, region_gate=None):
        super().__init__()
        self.base = base
        for p in self.base.parameters():
            p.requires_grad_(False)
        self.adapter = AdapterEdge(base.cfg.d_model, ADAPTER_RANK)
        # gate as a (1, V, 1) float buffer for broadcast over (B, V, T)
        if region_gate is None:
            self.register_buffer("gate", None, persistent=False)
        else:
            self.register_buffer(
                "gate", region_gate.view(1, -1, 1).float(), persistent=False)

    def trainable_params(self):
        return list(self.adapter.parameters())

    def n_trainable(self):
        return sum(p.numel() for p in self.trainable_params())

    def _trunk_norm_out(self, tokens):
        b = self.base
        x = b.embed(tokens).transpose(1, 2)
        x = b.embed_conv(x)
        for layer in b.trunk:
            x = layer(x)
        x, _ = b.moe(x)
        return b.norm_out(x)

    def forward(self, tokens, targets=None):
        h = self._trunk_norm_out(tokens)
        hp = self.adapter(h)
        logits_adapted = self.base.readout(hp)
        if self.gate is None:
            logits = logits_adapted
        else:
            logits_base = self.base.readout(h)        # FROZEN-readout pass-through
            logits = self.gate * logits_adapted + (1.0 - self.gate) * logits_base
        out = {"logits": logits}
        if targets is not None:
            ce = F.cross_entropy(
                logits.transpose(1, 2).reshape(-1, self.base.cfg.vocab_size),
                targets.reshape(-1),
            )
            out["ce_loss"] = ce
        return out


@torch.no_grad()
def eval_ce(model, batches):
    device = next(model.parameters()).device
    ces = []
    for toks, tgts in batches:
        toks, tgts = toks.to(device), tgts.to(device)
        out = model(toks, tgts)
        ces.append(out["ce_loss"].item())
    t = torch.tensor(ces)
    return t.mean().item(), t.std(unbiased=False).item()


def infer_config_and_load(ckpt_path):
    obj = torch.load(ckpt_path, map_location="cpu")
    if isinstance(obj, dict) and "state_dict" in obj:
        sd = obj["state_dict"]
        meta = obj.get("config") or obj.get("cfg") or {}
    elif isinstance(obj, dict) and any(k.startswith(("embed", "trunk", "moe", "readout")) for k in obj):
        sd = obj
        meta = {}
    else:
        sd = obj.get("model", obj) if isinstance(obj, dict) else obj
        meta = obj.get("config", {}) if isinstance(obj, dict) else {}
    d_model = sd["embed.weight"].shape[1]
    vocab = sd["embed.weight"].shape[0]
    n_trunk = len({k.split(".")[1] for k in sd if k.startswith("trunk.")})
    n_experts = len({k.split(".")[2] for k in sd if k.startswith("moe.experts.")})
    variant = meta.get("variant", "AB") if isinstance(meta, dict) else "AB"
    cfg = CLMConfig(vocab_size=vocab, d_model=d_model, n_trunk_layers=n_trunk,
                    n_experts=n_experts, variant=variant)
    base = CLMConvMoE(cfg)
    base.load_state_dict(sd, strict=False)
    return base, cfg, sd


def fresh_base(sd, cfg):
    base = CLMConvMoE(cfg)
    base.load_state_dict(sd, strict=False)
    return base


def run_arm(sd, cfg, gated, region_gate, device, seed):
    """One adaptation arm. Same data/seed for both arms; only `gated` differs."""
    torch.manual_seed(seed)
    base = fresh_base(sd, cfg).to(device)
    model = RegionGatedCLM(base, region_gate if gated else None).to(device)

    base_bytes = make_lane_bytes("web", 30000, BASE_SEED)
    new_bytes = make_lane_bytes("new", 30000, NEW_SEED)
    half = len(new_bytes) // 2
    base_eval = batchify(base_bytes, SEQ_LEN, N_EVAL_BATCHES, BATCH_SIZE, BASE_SEED)
    new_eval = batchify(new_bytes[half:], SEQ_LEN, N_EVAL_BATCHES, BATCH_SIZE, NEW_SEED + 1)
    new_train = batchify(new_bytes[:half], SEQ_LEN, N_ADAPT, BATCH_SIZE, NEW_SEED + 2)

    model.eval()
    ce_base_pre, sd_base_pre = eval_ce(model, base_eval)
    ce_new_pre, _ = eval_ce(model, new_eval)

    opt = torch.optim.Adam(model.trainable_params(), lr=LR_EDGE)
    model.train()
    for i in range(N_ADAPT):
        toks, tgts = new_train[i]
        toks, tgts = toks.to(device), tgts.to(device)
        out = model(toks, tgts)
        opt.zero_grad()
        out["ce_loss"].backward()
        opt.step()
    model.eval()
    ce_base_post, _ = eval_ce(model, base_eval)
    ce_new_post, _ = eval_ce(model, new_eval)

    z_drop = (ce_base_post - ce_base_pre) / max(sd_base_pre, 1e-6)
    gain = ce_new_pre - ce_new_post
    return {
        "arm": "gated" if gated else "ungated",
        "n_trainable_params": model.n_trainable(),
        "ce_base_pre": round(ce_base_pre, 5), "sd_base_pre": round(sd_base_pre, 5),
        "ce_base_post": round(ce_base_post, 5),
        "ce_new_pre": round(ce_new_pre, 5), "ce_new_post": round(ce_new_post, 5),
        "z_drop": round(z_drop, 5), "gain": round(gain, 5),
    }


def run_seed(sd, cfg, device, seed):
    new_bytes = make_lane_bytes("new", 30000, NEW_SEED)
    new_train = batchify(new_bytes[:len(new_bytes) // 2], SEQ_LEN, N_ADAPT, BATCH_SIZE, NEW_SEED + 2)
    gate = derive_region(new_train, cfg.vocab_size, REGION_MASS)
    r_new_size = int(gate.sum().item())

    ungated = run_arm(sd, cfg, False, None, device, seed)
    gated = run_arm(sd, cfg, True, gate.to(device), device, seed)

    retain = bool(gated["z_drop"] < ungated["z_drop"])
    gain_keep = bool(gated["gain"] >= ungated["gain"])
    return {
        "seed": seed,
        "region_R_new_size": r_new_size, "region_R_base_size": cfg.vocab_size - r_new_size,
        "ungated": ungated, "gated": gated,
        "RETAIN_pass": retain, "GAIN_pass": gain_keep,
        "verdict": "GREEN" if (retain and gain_keep) else "RED",
    }


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--ckpt", required=True)
    ap.add_argument("--out", required=True)
    ap.add_argument("--seeds", default="", help="comma list; default = registered sweep")
    args = ap.parse_args()
    device = "cuda" if torch.cuda.is_available() else "cpu"
    os.makedirs(args.out, exist_ok=True)

    base, cfg, sd = infer_config_and_load(args.ckpt)
    print("[h882] backbone d=%d L=%d E=%d V=%d params=%d device=%s" % (
        cfg.d_model, cfg.n_trunk_layers, cfg.n_experts, cfg.vocab_size,
        base.num_params(), device))

    seeds = ([int(x) for x in args.seeds.split(",") if x.strip()]
             if args.seeds.strip() else SEED_SWEEP)

    runs = []
    for s in seeds:
        r = run_seed(sd, cfg, device, s)
        runs.append(r)
        print("[h882 seed=%d R_new=%d] ungated z_drop=%.5f gain=%.5f | gated z_drop=%.5f gain=%.5f | RETAIN=%s GAIN=%s -> %s" % (
            r["seed"], r["region_R_new_size"],
            r["ungated"]["z_drop"], r["ungated"]["gain"],
            r["gated"]["z_drop"], r["gated"]["gain"],
            r["RETAIN_pass"], r["GAIN_pass"], r["verdict"]))

    head = next((r for r in runs if r["seed"] == RUN_SEED), runs[0])
    n_green = sum(1 for r in runs if r["verdict"] == "GREEN")
    majority = "GREEN" if n_green * 2 > len(runs) else "RED"

    result = {
        "campaign": "H_882 region-gated plasticity (영역별 학습 게이트 · BOUND/interference E5)",
        "prereg_commit": "54a2bae7d",
        "rung": "mid d512/L8/E8",
        "adapter_rank": ADAPTER_RANK,
        "region_mass_coverage": REGION_MASS,
        "edge_learn": "SW-sim (H_679 HW edge-learn real); measurement rung a_scale_honest_scope",
        "gate_def": "R_new = byte set covering >=99% of new-context TRAIN target mass; adapter logit-contribution masked to R_new; R_base = FROZEN-readout pass-through",
        "falsifier": "RETAIN: z_drop_gated < z_drop_ungated  AND  GAIN: gain_gated >= gain_ungated",
        "headline_seed": RUN_SEED,
        "headline": head,
        "runs": runs,
        "n_green": n_green, "n_runs": len(runs), "majority": majority,
        "verdict": head["verdict"],
        "note": "two arms identical except the region gate (E5); thresholds prereg 54a2bae7d verbatim, post-tuning 0; 🟢 iff both gates pass else 🔴 (a_paper_negative_ok)",
    }
    with open(os.path.join(args.out, "clm_region_gated_result.json"), "w") as f:
        json.dump(result, f, indent=2)
    print("[h882] headline_seed=%d verdict=%s | majority=%s (%d/%d GREEN)" % (
        RUN_SEED, result["verdict"], majority, n_green, len(runs)))
    print("=== H882 COMPLETE ===")


if __name__ == "__main__":
    main()
