#!/usr/bin/env python3
"""H_9124 lever#1 — derivation-trace procedural corpus generator (2-arm control).

Emits TWO byte corpora that are CONTENT-MATCHED and differ ONLY in target format:
  DERIV : composite prompt -> explicit derivation trace -> final OUT passage
  FLAT  : composite prompt -> final OUT passage only   (census#3 coverage-flat homolog)

Both arms are generated from the SAME instance stream (same pairs, same keyword
fills, same RNG order); the FLAT sample is literally the DERIV sample with the
derivation middle removed. Only data-format is varied (the controlled variable).

Frozen-bar alignment (cli/evaluate.py g_eval_g1 + core/g6_ideation.py):
  concept seeds cz[0..4]  (must NOT appear as a composite in training for the
  4 TESTED prefix composites -> memorization-free generalization test):
    c0 consciousness arises from cells
    c1 tension ripples between distant minds
    c2 memory composes into new meaning
    c3 silence still carries information
    c4 the engine dreams when alone
  coverage families (any 1 exact word hits the family):
    f0 consciousness|cells|mind|aware
    f1 tension|ripple|distant|between
    f2 memory|meaning|compose|new
    f3 silence|information|quiet|carries
    f4 dream|engine|alone|sleep

HELD-OUT (never shown in training, in any order):
  - the pair {c0,c1}          -> tests k=2 prefix composite (pair generalization)
  - ALL triples/higher (arity>=3 never trained; only pairs are)
                              -> tests k=3,4,5 prefix composites (arity gen.)
"""
import random, sys

SEED = 7
random.seed(SEED)

# concept seed sentences (EXACTLY cz[] from g6_ideation._g6_concepts)
S = [
    "consciousness arises from cells",
    "tension ripples between distant minds",
    "memory composes into new meaning",
    "silence still carries information",
    "the engine dreams when alone",
]
# per-family EXACT detector words (real English -> kwr-safe; each hits its family)
KW = [
    ["consciousness", "cells", "mind", "aware"],
    ["tension", "distant", "between"],
    ["memory", "meaning", "new"],
    ["silence", "information", "quiet", "carries"],
    ["engine", "alone", "dream"],
]

# training pairs = all unordered pairs EXCEPT the held-out {0,1}, both orderings.
HELDOUT_PAIR = frozenset((0, 1))
train_pairs = []
for i in range(5):
    for j in range(5):
        if i == j:
            continue
        if frozenset((i, j)) == HELDOUT_PAIR:
            continue
        train_pairs.append((i, j))

DERIVE_LEAD = ["derive", "steps", "unfold", "trace"]
BIND = ["bind", "join", "weave", "link"]
CLOSE = ["new meaning arises", "meaning composes anew",
         "a new whole arises", "they compose into meaning"]


def two(fam):
    ks = KW[fam][:]
    random.shuffle(ks)
    return ks[0], ks[1 % len(ks)]


def instance(i, j):
    """Return (deriv_text, flat_text) for ordered concept pair (i,j)."""
    a1, a2 = two(i)
    b1, b2 = two(j)
    prompt = f"{S[i]}. {S[j]}. "
    out = f"out: {a1} {a2} meet {b1} {b2}, {random.choice(CLOSE)}.\n"
    deriv_mid = (f"{random.choice(DERIVE_LEAD)}: take {a1} and {a2}; "
                 f"take {b1} and {b2}; {random.choice(BIND)} {a1} with {b1}. ")
    return prompt + deriv_mid + out, prompt + out


def single(i):
    """Family-local single-concept passage (identical in both arms).
    Keeps single-concept coverage at 1 family so the G1 bar (cov>max_single)
    stays reachable."""
    a1, a2 = two(i)
    return f"{S[i]}. here {a1} and {a2} stand alone; {a1} holds {a2}.\n"


def build(n_comp_per_pair, n_single_per_concept):
    deriv, flat = [], []
    stream = []
    for _ in range(n_comp_per_pair):
        for (i, j) in train_pairs:
            stream.append(("comp", i, j))
    for _ in range(n_single_per_concept):
        for i in range(5):
            stream.append(("sing", i, None))
    random.shuffle(stream)
    for kind, i, j in stream:
        if kind == "comp":
            d, f = instance(i, j)
            deriv.append(d)
            flat.append(f)
        else:
            s = single(i)
            deriv.append(s)
            flat.append(s)
    return "".join(deriv), "".join(flat)


if __name__ == "__main__":
    outdir = sys.argv[1] if len(sys.argv) > 1 else "."
    # ~280 comp variants/pair * 18 pairs ~= 5000 comp docs; singles ~ 1500
    deriv, flat = build(n_comp_per_pair=280, n_single_per_concept=300)
    with open(f"{outdir}/deriv.txt", "w") as fh:
        fh.write(deriv)
    with open(f"{outdir}/flat.txt", "w") as fh:
        fh.write(flat)
    print(f"train_pairs={len(train_pairs)} (held-out {{0,1}})")
    print(f"DERIV bytes={len(deriv.encode())} FLAT bytes={len(flat.encode())}")
    print("--- DERIV sample head ---")
    print(deriv[:600])
    print("--- FLAT sample head ---")
    print(flat[:400])
