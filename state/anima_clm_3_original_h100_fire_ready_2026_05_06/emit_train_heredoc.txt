# === Step 2: train heredoc (run on remote pod) ===

ssh -p ${POD_PORT} root@${POD_IP} bash -se <<'TRAIN_SCRIPT'
set -euo pipefail

# --- environment setup ---
cd /workspace
git clone --depth 1 https://github.com/need-singularity/anima /workspace/anima 2>/dev/null || \
  (cd /workspace/anima && git fetch --depth 1 origin main && git reset --hard origin/main)
cd /workspace/anima

# --- Python venv + deps (RTX 5070 sm_120 lesson: torch 2.11.0+cu128) ---
python -m pip install --quiet --upgrade pip
python -m pip install --quiet \
  torch==2.11.0 \
  transformers==4.57.6 \
  peft \
  accelerate \
  datasets \
  numpy \
  einops

# --- HF token (from env injected at pod create) ---
export HF_HUB_ENABLE_HF_TRANSFER=1
[ -n "${HF_TOKEN:-}" ] && python -c "from huggingface_hub import login; import os; login(os.environ['HF_TOKEN'])"

# --- training launch (3-phase 100K steps, byte 256, 32 cells, 19 phi-boost) ---
mkdir -p /workspace/anima/runs/clm3-original-byte-55m

python ready/training/train_clm.py \
  --vocab-size 256 \
  --max-cells 32 \
  --d-model 768 \
  --ffn-dim 1536 \
  --n-layers 12 \
  --n-heads 12 \
  --context-len 1024 \
  --steps 100000 \
  --phase-mitosis 0:20000 \
  --phase-language 20000:60000 \
  --phase-combined 60000:100000 \
  --fibonacci-growth 1,1,2,3,5,8,13,21,32 \
  --phi-boost-techniques COMBO2,FX2,WI1,PX4,PX8,GD18,GD15,CL8,CL5,DD3,DD11,DD18,DD5,TL13,TL1,NV7,BV1,EV3,SC2 \
  --corpus data/corpus_mix_70wiki_30dialogue.txt \
  --output-dir /workspace/anima/runs/clm3-original-byte-55m \
  --falsifier-eval-every 10000 \
  2>&1 | tee /workspace/anima/runs/clm3-original-byte-55m/train.log

# --- post-train: HF private push (PRIVATE first per own 15) ---
python -c "
from huggingface_hub import HfApi
api = HfApi()
api.create_repo('need-singularity/anima-clm-3-original-byte-55m', private=True, exist_ok=True)
api.upload_folder(folder_path='/workspace/anima/runs/clm3-original-byte-55m', repo_id='need-singularity/anima-clm-3-original-byte-55m', repo_type='model')
"

echo '[train] DONE — verdict to be assigned anima-side after F-CLM3-orig-1..5 eval'
TRAIN_SCRIPT
