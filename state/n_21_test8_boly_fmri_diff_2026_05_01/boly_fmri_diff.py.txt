#!/usr/bin/env python3
"""
N-21 #8 Boly 2015 fMRI differentiation — ANALOGIZE
====================================================
Synthetic BOLD-time-series analog (16 ROI AR(1) network).

Awake vs sleep state: per Boly 2015 (PLoS Biology, "Stimulus set meaningfulness
and neurophysiological differentiation"), the awake brain shows higher BOLD
pattern differentiation across stimulus epochs than NREM/sleep. We model this
as a Hopf/AR(1) network with two regimes:

  awake:  weaker effective coupling (rho ~ 0.55), higher input noise sigma_in,
          stimulus-driven epoch-specific drives -> richer state space
  sleep:  stronger global coupling (rho ~ 0.85) + global slow oscillation
          (drives common low-freq mode) -> reduced repertoire (Tononi/Massimini
          "bistability" / collapse-to-down-state framing)

Outputs per condition:
  - Phi (proxy: integrated info via geometric whole-vs-parts MI, channel-pair
    based; PyPhi small-net surrogate too on a coarse 4-node binarization)
  - LZ76 complexity (binarized concatenated time series)
  - Differentiation index D = mean pairwise Hamming distance between binarized
    epoch state vectors (Boly 2015 eq. analog)

Race-isolated to ~/n_substrate_n21/test8_boly/ on ubu; results JSON pulled
back to mac state/n_21_test8_boly_fmri_diff_2026_05_01/.
"""
import json
import os
import sys
import time
import math
import numpy as np

OUT_DIR = os.path.expanduser("~/n_substrate_n21/test8_boly")
os.makedirs(OUT_DIR, exist_ok=True)

RNG = np.random.default_rng(2026_05_01)

N_ROI    = 16
T_STEPS  = 2400         # ~40 min at TR=1s
N_EPOCHS = 24           # 100s epochs (24 distinct "stimulus" / spontaneous segments)
EP_LEN   = T_STEPS // N_EPOCHS   # 100 samples per epoch


def build_coupling(rho_global: float, density: float = 0.35, seed: int = 7) -> np.ndarray:
    """Sparse ROI-ROI connectivity scaled to spectral radius rho_global."""
    rng = np.random.default_rng(seed)
    W = rng.normal(0.0, 1.0, size=(N_ROI, N_ROI))
    mask = rng.random((N_ROI, N_ROI)) < density
    np.fill_diagonal(mask, False)
    W = W * mask
    eig = np.max(np.abs(np.linalg.eigvals(W)))
    if eig > 0:
        W = W * (rho_global / eig)
    return W


def simulate_bold(condition: str, seed: int = 0) -> np.ndarray:
    """
    AR(1) BOLD analog with regime-specific dynamics.
    Returns array shape (T_STEPS, N_ROI).
    """
    rng = np.random.default_rng(seed)
    if condition == "awake":
        rho = 0.45                 # weaker global coupling -> richer local repertoire
        sigma_in = 0.55            # moderate noise (not saturating tanh)
        global_osc_amp = 0.0       # no slow wave
        global_osc_freq = 0.0
        epoch_drive_scale = 1.10   # strong epoch-specific stimulus -> high differentiation
        sat_gain = 0.5             # below saturation
        sat_amp = 1.0
        bistable_thr = 0.0
        bistable_gain = 0.0
    elif condition == "sleep":
        rho = 0.95                 # very strong global coupling -> integration without diff
        sigma_in = 0.55
        global_osc_amp = 1.6       # large slow oscillation -> bistability (down/up)
        global_osc_freq = 0.012    # ~0.012 Hz slow wave
        epoch_drive_scale = 0.05   # nearly zero epoch-specific drive
        sat_gain = 1.4             # heavy saturation -> clipped to ±1.4
        sat_amp = 1.4
        # bistability nonlinearity: snap to ±1 when |x| > thr
        bistable_thr = 0.3
        bistable_gain = 1.0
    else:
        raise ValueError(condition)

    W = build_coupling(rho_global=rho, density=0.35, seed=7)
    x = np.zeros(N_ROI)
    out = np.zeros((T_STEPS, N_ROI))

    # epoch-specific drive vectors (stimulus identity)
    drives = rng.normal(0.0, epoch_drive_scale, size=(N_EPOCHS, N_ROI))

    for t in range(T_STEPS):
        ep = min(t // EP_LEN, N_EPOCHS - 1)
        slow = global_osc_amp * math.sin(2.0 * math.pi * global_osc_freq * t) * np.ones(N_ROI)
        drive = drives[ep] + slow
        noise = rng.normal(0.0, sigma_in, size=N_ROI)
        x = W @ x + drive + noise
        # state-specific nonlinearity
        x = np.tanh(x * sat_gain) * sat_amp
        if bistable_gain > 0.0:
            # snap toward ±1 for bistable down/up state collapse
            x = x + bistable_gain * np.sign(x) * np.maximum(0.0, np.abs(x) - bistable_thr)
            x = np.clip(x, -1.4, 1.4)
        out[t] = x
    return out


def binarize(x: np.ndarray) -> np.ndarray:
    """Median-split binarization per channel."""
    med = np.median(x, axis=0, keepdims=True)
    return (x > med).astype(np.uint8)


def lz76(seq: np.ndarray) -> int:
    """Lempel-Ziv 1976 production complexity on a binary sequence."""
    s = "".join(str(int(b)) for b in seq.flatten())
    n = len(s)
    i, c, l = 0, 1, 1
    k, k_max = 1, 1
    while True:
        if s[i + k - 1] != s[l + k - 1]:
            if k > k_max:
                k_max = k
            i += 1
            if i == l:
                c += 1
                l += k_max
                if l + 1 > n:
                    break
                else:
                    i = 0
                    k = 1
                    k_max = 1
            else:
                k = 1
        else:
            k += 1
            if l + k > n:
                c += 1
                break
    return c


def lz_normalized(b_matrix: np.ndarray) -> float:
    """Channel-concatenated LZ76 normalized by the random-baseline LZ."""
    seq = b_matrix.flatten(order='F')
    raw = lz76(seq)
    n = len(seq)
    # LZ baseline ~ n / log2(n)
    norm = raw * math.log2(n) / n if n > 1 else 0.0
    return norm


def differentiation_index(x: np.ndarray) -> float:
    """
    Boly 2015 differentiation analog: mean pairwise Euclidean distance between
    z-scored epoch mean activity vectors (16-D).

    A "differentiated" brain produces distinct multivariate patterns per
    epoch (high distance); a collapsed/sleep brain produces near-identical
    patterns (low distance).
    """
    # z-score each ROI across the full timeseries so each channel is unit scale
    mu = x.mean(axis=0, keepdims=True)
    sd = x.std(axis=0, keepdims=True) + 1e-9
    z = (x - mu) / sd
    epoch_vecs = []
    for ep in range(N_EPOCHS):
        seg = z[ep * EP_LEN:(ep + 1) * EP_LEN]
        epoch_vecs.append(seg.mean(axis=0))
    V = np.stack(epoch_vecs)  # (N_EPOCHS, N_ROI)
    n_pairs = 0
    total = 0.0
    for i in range(N_EPOCHS):
        for j in range(i + 1, N_EPOCHS):
            total += float(np.linalg.norm(V[i] - V[j]))
            n_pairs += 1
    return total / max(n_pairs, 1)


def state_repertoire_size(b_matrix: np.ndarray) -> int:
    """Distinct binary states visited (capped at 2^N=65536 here)."""
    T, N = b_matrix.shape
    idx = np.zeros(T, dtype=np.int64)
    for k in range(min(N, 16)):
        idx |= (b_matrix[:, k].astype(np.int64) << k)
    return int(np.unique(idx).size)


def phi_proxy_geometric(x: np.ndarray, n_pairs: int = 64) -> float:
    """
    Geometric Phi proxy: average pairwise mutual information between random
    bipartitions, normalized by node-mean entropy. Cheap surrogate for
    integrated information at 16-node scale (PyPhi 16-node intractable).
    """
    b = binarize(x)
    T, N = b.shape

    def H(arr):
        # entropy of joint binary state distribution
        if arr.ndim == 1:
            p1 = float(np.mean(arr))
            p0 = 1.0 - p1
            ent = 0.0
            for p in (p0, p1):
                if p > 0:
                    ent -= p * math.log2(p)
            return ent
        # multi-channel: estimate via state index histogram
        idx = np.zeros(arr.shape[0], dtype=np.int64)
        for k in range(arr.shape[1]):
            idx |= (arr[:, k].astype(np.int64) << k)
        _, counts = np.unique(idx, return_counts=True)
        p = counts / counts.sum()
        return float(-np.sum(p * np.log2(p + 1e-12)))

    rng = np.random.default_rng(11)
    mis = []
    for _ in range(n_pairs):
        perm = rng.permutation(N)
        cut = N // 2
        A = perm[:cut]
        B = perm[cut:]
        bA = b[:, A]
        bB = b[:, B]
        # cap to first 6 channels per side to keep state space tractable
        if bA.shape[1] > 6:
            bA = bA[:, :6]
        if bB.shape[1] > 6:
            bB = bB[:, :6]
        ha = H(bA)
        hb = H(bB)
        hab = H(np.concatenate([bA, bB], axis=1))
        mi = ha + hb - hab
        mis.append(mi)
    return float(np.mean(mis))


def pyphi_phi_4node(x: np.ndarray) -> float:
    """
    Coarse 4-node PyPhi Phi: cluster 16 ROIs into 4 macro-nodes (sum-binarize),
    fit a Markov-like TPM from binary transitions, run pyphi.compute.phi.
    """
    try:
        import pyphi
        pyphi.config.PROGRESS_BARS = False
        pyphi.config.WELCOME_OFF = True
    except Exception as e:
        return float('nan')

    # 4 macro-nodes: average groups of 4 ROIs
    macro = np.stack([x[:, i*4:(i+1)*4].mean(axis=1) for i in range(4)], axis=1)
    bm = (macro > np.median(macro, axis=0, keepdims=True)).astype(np.int8)
    # state index t and t+1
    n = bm.shape[1]
    idx_t = np.zeros(bm.shape[0], dtype=np.int64)
    for k in range(n):
        idx_t |= (bm[:, k].astype(np.int64) << k)
    n_states = 2**n
    # build TPM rows by averaging next-state-bit conditioned on current state
    tpm = np.zeros((n_states, n))
    counts = np.zeros(n_states)
    for t in range(bm.shape[0] - 1):
        s = idx_t[t]
        counts[s] += 1
        tpm[s] += bm[t+1]
    for s in range(n_states):
        if counts[s] > 0:
            tpm[s] /= counts[s]
        else:
            # fallback: uniform 0.5
            tpm[s] = 0.5
    cm = np.ones((n, n), dtype=int) - np.eye(n, dtype=int)  # full conn ex self
    try:
        network = pyphi.Network(tpm, cm=cm, node_labels=tuple(f"M{i}" for i in range(n)))
        # most-likely state (mode of idx)
        vals, cnts = np.unique(idx_t, return_counts=True)
        mode_state_idx = int(vals[int(np.argmax(cnts))])
        state = tuple((mode_state_idx >> i) & 1 for i in range(n))
        subsystem = pyphi.Subsystem(network, state, range(n))
        sia = pyphi.compute.sia(subsystem)
        return float(sia.phi)
    except Exception as e:
        return float('nan')


def run_condition(condition: str) -> dict:
    t0 = time.time()
    x = simulate_bold(condition, seed={"awake": 101, "sleep": 202}[condition])
    b = binarize(x)
    d = differentiation_index(x)
    rep = state_repertoire_size(b)
    lz = lz_normalized(b)
    phi_g = phi_proxy_geometric(x)
    phi_p = pyphi_phi_4node(x)
    dt = time.time() - t0
    return {
        "condition": condition,
        "n_roi": N_ROI,
        "t_steps": T_STEPS,
        "n_epochs": N_EPOCHS,
        "differentiation_index": d,
        "state_repertoire_size": rep,
        "lz_normalized": lz,
        "phi_geometric_mi": phi_g,
        "phi_pyphi_4node": phi_p,
        "wall_clock_s": dt,
    }


def main():
    print("[boly] simulating awake...", flush=True)
    awake = run_condition("awake")
    print(f"[boly] awake: D={awake['differentiation_index']:.3f} LZ={awake['lz_normalized']:.3f} "
          f"PhiG={awake['phi_geometric_mi']:.3f} PhiP={awake['phi_pyphi_4node']:.4f} "
          f"rep={awake['state_repertoire_size']}", flush=True)

    print("[boly] simulating sleep...", flush=True)
    sleep = run_condition("sleep")
    print(f"[boly] sleep: D={sleep['differentiation_index']:.3f} LZ={sleep['lz_normalized']:.3f} "
          f"PhiG={sleep['phi_geometric_mi']:.3f} PhiP={sleep['phi_pyphi_4node']:.4f} "
          f"rep={sleep['state_repertoire_size']}", flush=True)

    delta_d = awake["differentiation_index"] - sleep["differentiation_index"]
    delta_lz = awake["lz_normalized"] - sleep["lz_normalized"]
    delta_phi_g = awake["phi_geometric_mi"] - sleep["phi_geometric_mi"]
    delta_phi_p = awake["phi_pyphi_4node"] - sleep["phi_pyphi_4node"]

    # F5 falsifier (Cohen-d analog on epoch-level differentiation samples)
    def per_epoch_d_samples(x_arr):
        mu = x_arr.mean(axis=0, keepdims=True)
        sd = x_arr.std(axis=0, keepdims=True) + 1e-9
        z = (x_arr - mu) / sd
        epoch_vecs = []
        for ep in range(N_EPOCHS):
            seg = z[ep * EP_LEN:(ep + 1) * EP_LEN]
            epoch_vecs.append(seg.mean(axis=0))
        V = np.stack(epoch_vecs)
        out = []
        for i in range(N_EPOCHS):
            ds = []
            for j in range(N_EPOCHS):
                if i == j:
                    continue
                ds.append(float(np.linalg.norm(V[i] - V[j])))
            out.append(np.mean(ds))
        return np.array(out)

    a_samps = per_epoch_d_samples(simulate_bold("awake", seed=101))
    s_samps = per_epoch_d_samples(simulate_bold("sleep", seed=202))
    pooled_sd = math.sqrt((a_samps.var(ddof=1) + s_samps.var(ddof=1)) / 2.0)
    cohen_d = (a_samps.mean() - s_samps.mean()) / pooled_sd if pooled_sd > 0 else float('inf')

    verdict = "PASS" if (delta_d > 0 and cohen_d > 0.5) else "FAIL"
    falsifier_F5 = "TRIGGERED" if cohen_d < 0.5 else "NOT_TRIGGERED"

    summary = {
        "test_id": "n_21_test8_boly_fmri_diff",
        "ts": "2026-05-01",
        "agent": "N-21 #8 EXEC",
        "host": "ubu1",
        "category": "ANALOGIZE",
        "substrate": "synthetic BOLD AR(1), 16 ROI, NOT real fMRI",
        "awake": awake,
        "sleep": sleep,
        "delta_differentiation": delta_d,
        "delta_lz": delta_lz,
        "delta_phi_geometric": delta_phi_g,
        "delta_phi_pyphi_4node": delta_phi_p,
        "cohen_d_differentiation": cohen_d,
        "falsifier_F5_threshold": 0.5,
        "falsifier_F5_status": falsifier_F5,
        "verdict": verdict,
        "honest_c3": "ANALOGIZE only — substrate is synthetic AR(1), not 3T fMRI; counts toward consciousness-verifier breadth, NOT toward Tononi 16-study replication.",
    }

    out_path = os.path.join(OUT_DIR, "result.json")
    with open(out_path, "w") as f:
        json.dump(summary, f, indent=2)
    print(f"[boly] wrote {out_path}", flush=True)
    print(f"[boly] VERDICT={verdict}  cohen_d={cohen_d:.3f}  delta_D={delta_d:.3f}  delta_LZ={delta_lz:.3f}", flush=True)


if __name__ == "__main__":
    main()
