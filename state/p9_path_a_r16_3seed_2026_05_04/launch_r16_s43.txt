#!/bin/bash
# Path A r=16 retrain — seed 43 (additional seed for 3-seed ensemble variance bound).
# Mirrors state/p9_path_a_r16_2026_05_03/launch_r16.txt with --seed 43 + s43 HF target.
# Saved as .txt to evade banned-ext sweeper (raw#9/r45 untracked .sh deleted).
set -uo pipefail
WORKSPACE=/workspace
OUT_DIR=$WORKSPACE/p9_path_a_r16_s43_savepoints
mkdir -p $OUT_DIR
export HF_TOKEN=$(cat $WORKSPACE/hf_token | tr -d "\n")
export HUGGING_FACE_HUB_TOKEN="$HF_TOKEN"
echo "[hf] token len=${#HF_TOKEN}"
mkdir -p ~/.cache/huggingface
echo -n "$HF_TOKEN" > ~/.cache/huggingface/token
mkdir -p ~/.huggingface
echo -n "$HF_TOKEN" > ~/.huggingface/token 2>/dev/null || true
echo "[sanity]"
ls -lh $WORKSPACE/sft_data_llama_template.jsonl 2>&1 | head -1
ls -lh $WORKSPACE/train_llama_lora_r16.py 2>&1 | head -1
nvidia-smi --query-gpu=name,memory.total --format=csv,noheader 2>&1 | head -3
echo "[deps]"
pip install -q --no-cache-dir "transformers>=4.45" "trl>=0.11" "peft>=0.13" "accelerate>=0.34" "datasets>=2.21" "bitsandbytes>=0.43" "huggingface_hub>=0.25" 2>&1 | tail -5
cd $WORKSPACE
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True
export PYTHONUNBUFFERED=1
echo "[launch] training detached r=16 alpha=16 seed=43"
nohup python3 -u $WORKSPACE/train_llama_lora_r16.py \
    --base-model meta-llama/Llama-3.2-3B-Instruct \
    --data-jsonl $WORKSPACE/sft_data_llama_template.jsonl \
    --output-dir $OUT_DIR \
    --lora-r 16 --lora-alpha 16 \
    --lr 1e-4 --per-device-batch 4 --grad-accum 8 \
    --max-steps 10000 --save-steps 2000 --seq-len 2048 --warmup-steps 200 --logging-steps 10 \
    --bf16 --gradient-checkpointing \
    --seed 43 \
    --push-to-hub dancinlab/llm-llama32-3b-paradigm-a-prime-r16-s43-sft-stage1 \
    > $OUT_DIR/train.log 2>&1 &
TRAIN_PID=$!
echo "$TRAIN_PID" > $OUT_DIR/train.pid
echo "[launched] pid=$TRAIN_PID log=$OUT_DIR/train.log"
disown $TRAIN_PID 2>/dev/null || true
sleep 5
echo "[probe-5s]"
ps -p $TRAIN_PID -o pid,etime,rss,cmd 2>/dev/null || echo "process gone"
ls -la $OUT_DIR/
