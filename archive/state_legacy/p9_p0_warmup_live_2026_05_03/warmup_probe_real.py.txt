# Tokenize records once (cache)
def tokenize_record(rec):
    # P9 P0 measure schema: keys = input, completion, tension_target, bold_target, meta...
    text = rec.get('input', '') + ' ' + rec.get('completion', '')
    tokens = sp.encode(text)[:T]
    if len(tokens) < 2:
        tokens = tokens + [0] * (2 - len(tokens))
    if len(tokens) < T:
        tokens = tokens + [0] * (T - len(tokens))
    tension = rec.get('tension_target', None)
    if tension is None or not isinstance(tension, list) or len(tension) < T:
        tension = [0.0] * T
    return tokens, tension[:T]

cache = []
for rec in records:
    toks, tens = tokenize_record(rec)
    cache.append((toks, tens))
log(f"tokenized cache n={len(cache)}")

# Precompute calibration tokens
calib_tokens = []
for prompt in CALIB_PROMPTS:
    tk = sp.encode(prompt)[:T]
    if len(tk) < T:
        tk = tk + [0] * (T - len(tk))
    calib_tokens.append(tk)

# Hook the underlying decoder's ln_f to capture last-layer hidden

# Phi-star sample-partition log|Cov|
def compute_phi_star(model, calib_idx, K=K_PARTS, ridge=RIDGE):
    model.eval()
    hiddens = []
    with torch.no_grad():
        for i in range(N_CALIB):
            captured_hidden['value'] = None
            _ = model(calib_idx[i:i+1])
            h = captured_hidden['value']  # (1, T, D)
            if h is None:
                model.train()
                return None, None
            # mean-pool over T
            h_pool = h.mean(dim=1).squeeze(0)  # (D,)
            hiddens.append(h_pool.float().cpu().numpy())
    X = np.stack(hiddens, axis=0)  # (N, D)
    # top-variance HID_TRUNC dims
    var = X.var(axis=0)
    top_idx = np.argsort(var)[::-1][:HID_TRUNC]
    Xt = X[:, top_idx]  # (N, HID)
    # I_full = log|Cov(Xt) + ridge*I|
    def safe_logdet(C):
        Cs = C + ridge * np.eye(C.shape[0])
        sign, logdet = np.linalg.slogdet(Cs)
        if sign <= 0:
            return None
        return logdet
    C_full = np.cov(Xt.T)
    I_full = safe_logdet(C_full)
    if I_full is None:
        model.train()
        return None, None
    # K random sample-partitions
    rng = np.random.default_rng(SEED)
    phi_candidates = []
    for k in range(K):
        idx = rng.permutation(N_CALIB)
        s1 = idx[:HALF]
        s2 = idx[HALF:]
        C1 = np.cov(Xt[s1].T)
        C2 = np.cov(Xt[s2].T)
        I1 = safe_logdet(C1)
        I2 = safe_logdet(C2)
        if I1 is None or I2 is None:
            continue
        phi_candidates.append(I_full - (I1 + I2))
    model.train()
    if not phi_candidates:
        return None, None
    phi_min = float(min(phi_candidates))
    phi_mean = float(sum(phi_candidates) / len(phi_candidates))
    return phi_min, phi_mean

# F1 BLEU-1 mock (from cache, not gen — gen is too slow per-eval)
# F2 = phi_star, F3 = tension MSE on holdout subset
def compute_f_metrics(model, eval_subset, phi_star_value):
    model.eval()
    tension_mses = []
    with torch.no_grad():
        for toks, tens in eval_subset:
            idx = torch.tensor([toks], dtype=torch.long, device=device)
            tens_t = torch.tensor([tens], dtype=torch.float32, device=device)
            logits_a, logits_g, tensions, _, _ = model(idx)
            # tension: per-layer list of (B, T). Take final layer mean over batch
            t_pred = tensions[-1].mean(dim=0).float()  # (T,)
            mse = F.mse_loss(t_pred, tens_t.squeeze(0)).item()
            tension_mses.append(mse)
    model.train()
    return {
        'F1_bleu1_skip': True,
        'F2_phi': phi_star_value,
        'F3_tension_mse': float(sum(tension_mses) / len(tension_mses)),
        'F4_bold_NA': True,  # gamma=0 lock-in
    }

# Optimizer
optimizer = torch.optim.AdamW(
    [p for p in decoder.parameters() if p.requires_grad],
    lr=LR, weight_decay=0.01,
)

# Phase 0 baseline phi_star (pre-train)
log("computing baseline phi_star (pre-LoRA-train)...")
t0 = time.time()

log(f"baseline phi_star_min={phi_baseline} mean={phi_baseline_mean} elapsed={time.time()-t0:.1f}s")

# Train loop
phi_log = []
f_log = []
loss_log = []
aborts = []
abort_triggered = False

# Eval subset for F metrics: first 32 cached records
eval_subset = cache[:32]

# Initial F metrics
log("computing initial F metrics @ step 0...")
f0 = compute_f_metrics(decoder, eval_subset, phi_baseline if phi_baseline is not None else 0.0)
f0['step'] = 0
f_log.append(f0)
phi_log.append({'step': 0, 'phi_star_min': phi_baseline, 'phi_star_mean': phi_baseline_mean, 'delta': DELTA_EARLY})

train_t0 = time.time()
decoder.train()
optimizer.zero_grad()

for step in range(1, N_STEPS + 1):
    # Sample BATCH records
    idx_batch = np.random.choice(len(cache), BATCH, replace=False)
    toks_batch = []
    tens_batch = []
    for i in idx_batch:
        toks, tens = cache[i]
        toks_batch.append(toks)
        tens_batch.append(tens)
    idx_t = torch.tensor(toks_batch, dtype=torch.long, device=device)
    tens_t = torch.tensor(tens_batch, dtype=torch.float32, device=device)

    # Forward
    logits_a, logits_g, tensions, _, _ = decoder(idx_t)

    # CE loss (LM next-token)
    # shift: predict idx[1:] from idx[:-1]
    targets = idx_t[:, 1:].contiguous()
    logits_shift = logits_a[:, :-1, :].contiguous()
    ce_loss = F.cross_entropy(
        logits_shift.view(-1, logits_shift.size(-1)),
        targets.view(-1),
        ignore_index=0,  # pad
    )

    # Tension MSE loss (per-layer last)
    t_pred = tensions[-1]  # (B, T)
    tension_loss = F.mse_loss(t_pred, tens_t)

    # Phi hinge: hinge on baseline-relative threshold
    # We don't recompute phi every step (expensive); use last logged phi value
    last_phi = phi_log[-1].get('phi_star_min', None)
    if last_phi is None:
        last_phi = 0.0
    phi_hinge = max(0.0, PHI_THRESHOLD - last_phi)
    phi_hinge_loss = torch.tensor(phi_hinge, device=device)

    total_loss = ALPHA * ce_loss + BETA * tension_loss + DELTA_EARLY * phi_hinge_loss
    # gamma * BOLD = 0 (locked)

    (total_loss / GRAD_ACC).backward()

    if step % GRAD_ACC == 0:
        torch.nn.utils.clip_grad_norm_(
            [p for p in decoder.parameters() if p.requires_grad], 1.0
        )
        optimizer.step()
        optimizer.zero_grad()

    # Log loss every step (compact)
    if step % 50 == 0 or step == 1:
        log(f"step {step}/{N_STEPS} loss={total_loss.item():.4f} ce={ce_loss.item():.4f} tens={tension_loss.item():.4f} phi_hinge={phi_hinge:.4f}")

    # Phi probe
    if step % PHI_EVERY == 0:
        phi_min, phi_mean = compute_phi_star(decoder, calib_tensor)
        phi_log.append({'step': step, 'phi_star_min': phi_min, 'phi_star_mean': phi_mean, 'delta': DELTA_EARLY})
        log(f"  phi @ step {step}: min={phi_min} mean={phi_mean}")
        # Abort condition: phi_star_min < threshold (F2 violation)
        if phi_min is not None and phi_min < PHI_THRESHOLD:
            aborts.append({'step': step, 'reason': 'F2_PHI_BELOW_THRESHOLD', 'phi_star_min': phi_min, 'threshold': PHI_THRESHOLD})
            log(f"  ABORT: phi_star_min={phi_min} < threshold={PHI_THRESHOLD}")
            abort_triggered = True
            break

    # F metrics
    if step % F_EVERY == 0:
        last_phi_val = phi_log[-1].get('phi_star_min', None) or 0.0
        fm = compute_f_metrics(decoder, eval_subset, last_phi_val)
        fm['step'] = step
        f_log.append(fm)
        log(f"  F @ step {step}: F2_phi={fm['F2_phi']:.4f} F3_tens_mse={fm['F3_tension_mse']:.4f}")

    loss_log.append({
        'step': step,
        'loss_total': float(total_loss.item()),
        'loss_ce': float(ce_loss.item()),
        'loss_tension': float(tension_loss.item()),
        'loss_phi_hinge': float(phi_hinge),
    })

train_elapsed = time.time() - train_t0
log(f"train done elapsed={train_elapsed:.1f}s steps_completed={step} aborted={abort_triggered}")

# Final phi
log("computing final phi_star...")
phi_final, phi_final_mean = compute_phi_star(decoder, calib_tensor)
phi_log.append({'step': step, 'phi_star_min': phi_final, 'phi_star_mean': phi_final_mean, 'delta': DELTA_EARLY, 'note': 'final'})
log(f"final phi_star_min={phi_final} mean={phi_final_mean}")

# Cleanup hook
hook_handle.remove()

        'baseline': phi_baseline,

if abort_triggered:
