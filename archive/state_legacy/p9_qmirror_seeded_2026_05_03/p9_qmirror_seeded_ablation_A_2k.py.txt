# qmirror QRNG — HMAC-DRBG SHA-256 seeded by IonQ Forte 1 quantum bits
# (Algorithm: NIST SP 800-90A Rev.1 §10.1.2; verified by cond.4 NIST tier-1+ 7/7 PASS)
# =========================================================================
class QmirrorHmacDrbgSha256:
    OUTLEN = 32

    def __init__(self, entropy: bytes, nonce: bytes, pers: bytes = b""):
        self.K = b"\x00" * self.OUTLEN
        self.V = b"\x01" * self.OUTLEN
        self._update(entropy + nonce + pers)
        self.reseed_counter = 1
        self.bytes_drawn = 0

    def _hmac(self, key, data):
        return hmac.new(key, data, hashlib.sha256).digest()

    def _update(self, provided_data: bytes):
        self.K = self._hmac(self.K, self.V + b"\x00" + provided_data)
        self.V = self._hmac(self.K, self.V)
        if provided_data:
            self.K = self._hmac(self.K, self.V + b"\x01" + provided_data)
            self.V = self._hmac(self.K, self.V)

    def generate(self, n_bytes: int) -> bytes:
        out = b""
        while len(out) < n_bytes:
            self.V = self._hmac(self.K, self.V)
            out += self.V
        self._update(b"")
        self.reseed_counter += 1
        self.bytes_drawn += n_bytes
        return out[:n_bytes]

    def get_uint64(self) -> int:
        return struct.unpack(">Q", self.generate(8))[0]

    def get_uint32(self) -> int:
        return struct.unpack(">I", self.generate(4))[0]


def build_qmirror_drbg():
    """Build the qmirror QRNG instance from the IonQ Forte 1 quantum entropy seed."""
    meta = json.loads(Path(QMIRROR_DRBG_META).read_text())
    raw = json.loads(Path(QMIRROR_QMEAS_RAW).read_text())
    entropy = bytes.fromhex(raw["raw_bits_4096_hex"])
    nonce = meta["nonce"].encode()
    pers = meta["personalization_string"].encode()
    drbg = QmirrorHmacDrbgSha256(entropy, nonce, pers)
    # Verify cond.4 chain-of-custody: re-derive first 1024 bytes and check vs HMAC-DRBG meta.
    expected = bytes.fromhex(meta["first_output_1024_bytes_hex"])
    first = drbg.generate(1024)
    if first != expected:
        raise RuntimeError("qmirror DRBG chain-of-custody MISMATCH (re-derivation differs from meta)")
    return drbg, meta, raw

    text = rec.get('input', '') + ' ' + rec.get('completion', '')

    tension = rec.get('tension_target', None)
    if tension is None or not isinstance(tension, list) or len(tension) < T:
        tension = [0.0] * T
    return tokens, tension[:T]

def compute_phi_star(model, calib_idx, K=K_PARTS, ridge=RIDGE):
    model.eval()
    hiddens = []
    with torch.no_grad():
        for i in range(N_CALIB):
            captured_hidden['value'] = None
            _ = model(calib_idx[i:i+1])
            h = captured_hidden['value']
            if h is None:
                model.train()
                return None, None
            h_pool = h.mean(dim=1).squeeze(0)
            hiddens.append(h_pool.float().cpu().numpy())
    X = np.stack(hiddens, axis=0)
    var = X.var(axis=0)
    top_idx = np.argsort(var)[::-1][:HID_TRUNC]
    Xt = X[:, top_idx]

    def safe_logdet(C):
        Cs = C + ridge * np.eye(C.shape[0])
        sign, logdet = np.linalg.slogdet(Cs)
        if sign <= 0:
            return None
        return logdet

    C_full = np.cov(Xt.T)
    I_full = safe_logdet(C_full)
    if I_full is None:
        model.train()
        return None, None
    # === PHI partition RNG: also qmirror-driven ===
    rng = np.random.default_rng(QMIRROR_SEED_CALIB_PHI)
    phi_candidates = []
    for k in range(K):
        idx = rng.permutation(N_CALIB)
        s1 = idx[:HALF]
        s2 = idx[HALF:]
        C1 = np.cov(Xt[s1].T)
        C2 = np.cov(Xt[s2].T)
        I1 = safe_logdet(C1)
        I2 = safe_logdet(C2)
        if I1 is None or I2 is None:
            continue
        phi_candidates.append(I_full - (I1 + I2))
    model.train()
    if not phi_candidates:
        return None, None
    phi_min = float(min(phi_candidates))
    phi_mean = float(sum(phi_candidates) / len(phi_candidates))
    return phi_min, phi_mean


def compute_f1_bleu1(model, holdout_subset):
    model.eval()
    bleu1_scores = []
    with torch.no_grad():
        for h in holdout_subset:
            inp_tok = h['input_tokens']
            ref_tok = h['reference_tokens']
            if not ref_tok:
                continue
            ctx = torch.tensor([inp_tok], dtype=torch.long, device=device)
            gen_tok = []
            for _ in range(F1_GEN_LEN):
                if ctx.size(1) >= T:
                    break
                logits_a, _, _, _, _ = model(ctx)
                next_id = int(logits_a[0, -1, :].argmax().item())
                gen_tok.append(next_id)
                ctx = torch.cat([ctx, torch.tensor([[next_id]], dtype=torch.long, device=device)], dim=1)
                if next_id == 0:
                    break
            if not gen_tok:
                continue
            ref_counts = {}
            for t in ref_tok:
                ref_counts[t] = ref_counts.get(t, 0) + 1
            hits = 0
            seen = {}
            for t in gen_tok:
                if t in ref_counts:
                    seen_c = seen.get(t, 0)
                    if seen_c < ref_counts[t]:
                        hits += 1
                        seen[t] = seen_c + 1
            bleu1 = hits / max(1, len(gen_tok))
            bleu1_scores.append(bleu1)
    model.train()
    if not bleu1_scores:
        return 0.0
    return float(sum(bleu1_scores) / len(bleu1_scores))


def compute_f_metrics(model, eval_subset, holdout_subset, phi_star_value):
    model.eval()
    tension_mses = []
    with torch.no_grad():
        for toks, tens in eval_subset:
            idx = torch.tensor([toks], dtype=torch.long, device=device)
            tens_t = torch.tensor([tens], dtype=torch.float32, device=device)
            logits_a, logits_g, tensions, _, _ = model(idx)
            t_pred = tensions[-1].mean(dim=0).float()
            mse = F.mse_loss(t_pred, tens_t.squeeze(0)).item()
            tension_mses.append(mse)
    model.train()
    f1 = compute_f1_bleu1(model, holdout_subset)
    return {
        'F1_bleu1': f1,
        'F2_phi': phi_star_value,
        'F3_tension_mse': float(sum(tension_mses) / len(tension_mses)),
        'F4_bold_NA': True,
    }

# === BatchSampler patch: per-step batch index draw uses qmirror QRNG ===
# Each batch needs BATCH unique indices in [0, len(cache)).  We reservoir-sample without replacement
# from qmirror uint32 stream.
def qmirror_batch_indices(n_cache: int, batch_size: int) -> np.ndarray:
    """Sample `batch_size` unique indices in [0, n_cache) using qmirror QRNG."""
    chosen = []
    seen = set()
    # rejection-sample uint32 → mod n_cache; collision rate << 1% for n_cache=50K, batch=4
    while len(chosen) < batch_size:
        u = qmirror_drbg.get_uint32()
        idx = u % n_cache
        if idx in seen:
            continue
        seen.add(idx)
        chosen.append(idx)
    return np.array(chosen, dtype=np.int64)

        (total_loss / GRAD_ACC).backward()

        if step % GRAD_ACC == 0:
            torch.nn.utils.clip_grad_norm_(
                [p for p in decoder.parameters() if p.requires_grad], 1.0
            )
            optimizer.step()
            optimizer.zero_grad()

        if step % 100 == 0 or step == 1:
            log(f"step {step}/{N_STEPS} loss={total_loss.item():.4f} ce={ce_loss.item():.4f} tens={tension_loss.item():.4f} phi_hinge={phi_hinge:.4f} α={alpha_now:.2f} δ={delta_now}")
            loss_log_compact.append({
                'step': step,
                'loss_total': float(total_loss.item()),
                'loss_ce': float(ce_loss.item()),
                'loss_tension': float(tension_loss.item()),
                'loss_phi_hinge': float(phi_hinge),
                'alpha': alpha_now,
                'delta': delta_now,
            })

        if step % PHI_EVERY == 0:
            phi_min, phi_mean = compute_phi_star(decoder, calib_tensor)
            phi_log.append({'step': step, 'phi_star_min': phi_min, 'phi_star_mean': phi_mean, 'delta': delta_now})
            log(f"  phi @ step {step}: min={phi_min} mean={phi_mean} δ={delta_now}")
            if phi_min is not None and phi_min < PHI_THRESHOLD:
                aborts.append({'step': step, 'reason': 'F2_PHI_BELOW_THRESHOLD', 'phi_star_min': phi_min, 'threshold': PHI_THRESHOLD})
                log(f"  ABORT: phi_star_min={phi_min} < threshold={PHI_THRESHOLD}")
                abort_triggered = True
                break

        if step % F_EVERY == 0:
            last_phi_val = phi_log[-1].get('phi_star_min', None) or 0.0
            fm = compute_f_metrics(decoder, eval_subset, holdout_cache, last_phi_val)
            fm['step'] = step
            f_log.append(fm)
            log(f"  F @ step {step}: F1_bleu1={fm['F1_bleu1']:.4f} F2_phi={fm['F2_phi']:.4f} F3_tens_mse={fm['F3_tension_mse']:.4f}")

        if step in SAVE_AT:
            sp_subdir = f"step_{step}"
            sd = save_lora_adapter(decoder, step, sp_subdir)
            log(f"  SAVEPOINT @ step {step} → {sd}")
            savepoint_log.append({'step': step, 'path': sd})

except KeyboardInterrupt:
    log(f"INTERRUPTED @ step {step}")
    aborts.append({'step': step, 'reason': 'KeyboardInterrupt'})
    abort_triggered = True
except Exception as e:
    log(f"EXCEPTION @ step {step}: {e}")
    log(traceback.format_exc())
    aborts.append({'step': step, 'reason': f'EXCEPTION: {e}'})
    abort_triggered = True

train_elapsed = time.time() - train_t0
qmirror_bytes_used_in_train = qmirror_drbg.bytes_drawn - qmirror_bytes_at_train_start
log(f"train done elapsed={train_elapsed:.1f}s steps_completed={step} aborted={abort_triggered}")
log(f"qmirror entropy used: total_bytes={qmirror_drbg.bytes_drawn} train_only_bytes={qmirror_bytes_used_in_train} reseed_counter={qmirror_drbg.reseed_counter}")

phi_final, phi_final_mean = compute_phi_star(decoder, calib_tensor)
phi_log.append({'step': step, 'phi_star_min': phi_final, 'phi_star_mean': phi_final_mean, 'delta': get_delta(step), 'note': 'final'})

f_final['step'] = step
f_final['note'] = 'final'
f_log.append(f_final)
