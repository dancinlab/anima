#!/usr/bin/env python3
# P9 P0 measure probe - CLM v4 tension extraction on ubu1
# scope: N=100 records subset, T=64 forward, tension scalar per token
# purpose: per-record latency measurement -> 50K projection
# additive only, destructive=0, sft_data.jsonl read-only
import json
import os
import sys
import time
import traceback

# ubu1 paths
ANIMA_HOME = "/home/aiden/anima"
CLM_CKPT = f"{ANIMA_HOME}/checkpoints/clm_v4_350m/scale_350m/best.pt"
TOKENIZER = "/tmp/tokenizer_64k_multilingual.model"
SFT_DATA = "/tmp/sft_data_subset_100.jsonl"  # to be scp'd
OUTPUT = "/tmp/p9_p0_probe_clm_v4_tension_100.jsonl"
LATENCY_LOG = "/tmp/p9_p0_probe_latency.json"

sys.path.insert(0, f"{ANIMA_HOME}/models")
sys.path.insert(0, f"{ANIMA_HOME}/training")
sys.path.insert(0, f"{ANIMA_HOME}/anima")  # for consciousness laws
sys.path.insert(0, f"{ANIMA_HOME}")

# T=64 from P9 spec
T = 64

print(f"[probe] start ts={time.strftime('%Y-%m-%dT%H:%M:%S')}", flush=True)
print(f"[probe] python={sys.version.split()[0]}", flush=True)

import torch
print(f"[probe] torch={torch.__version__} cuda={torch.cuda.is_available()}", flush=True)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"[probe] device={device}", flush=True)

# load tokenizer
try:
    import sentencepiece as spm
    sp = spm.SentencePieceProcessor(model_file=TOKENIZER)
    vocab_size = sp.get_piece_size()
    print(f"[probe] tokenizer loaded vocab={vocab_size}", flush=True)
except Exception as e:
    print(f"[probe] FAIL tokenizer: {e}", flush=True)
    sys.exit(2)

# load CLM v4 model
try:
    from conscious_decoder import ConsciousDecoderV2
    print(f"[probe] ConsciousDecoderV2 imported", flush=True)
except Exception as e:
    print(f"[probe] FAIL import: {e}", flush=True)
    sys.exit(3)

# 350m config from train_clm.py:249
scale_cfg = {
    'd_model': 768,
    'n_layer': 16,
    'n_head': 12,
    'n_kv_head': 4,
    'block_size': 512,  # match ckpt (heartbeat 350m phase)
    'consciousness_dim': 192,
}

try:
    decoder = ConsciousDecoderV2(
        vocab_size=vocab_size,
        d_model=scale_cfg['d_model'],
        n_head=scale_cfg['n_head'],
        n_layer=scale_cfg['n_layer'],
        block_size=scale_cfg['block_size'],
        n_kv_head=scale_cfg['n_kv_head'],
        consciousness_dim=scale_cfg['consciousness_dim'],
        dropout=0.0,  # eval mode
        gate_strength=0.001,
        n_ca_rules=8,
    )
    decoder = decoder.to(device)
    decoder.eval()
    n_params = sum(p.numel() for p in decoder.parameters())
    print(f"[probe] decoder built params={n_params:,}", flush=True)
except Exception as e:
    print(f"[probe] FAIL decoder build: {e}", flush=True)
    traceback.print_exc()
    sys.exit(4)

# load ckpt
try:
    t0 = time.time()
    ck = torch.load(CLM_CKPT, map_location=device, weights_only=False)
    sd = ck.get('decoder', ck) if isinstance(ck, dict) else ck
    missing, unexpected = decoder.load_state_dict(sd, strict=False)
    t_load = time.time() - t0
    print(f"[probe] ckpt loaded {t_load:.2f}s missing={len(missing)} unexpected={len(unexpected)}", flush=True)
    if len(missing) > 50:
        print(f"[probe] WARN many missing keys, e.g. {missing[:5]}", flush=True)
except Exception as e:
    print(f"[probe] FAIL ckpt load: {e}", flush=True)
    traceback.print_exc()
    sys.exit(5)

# load SFT subset
records = []
try:
    with open(SFT_DATA, 'r') as f:
        for line in f:
            records.append(json.loads(line))
    print(f"[probe] loaded {len(records)} records", flush=True)
except Exception as e:
    print(f"[probe] FAIL load sft: {e}", flush=True)
    sys.exit(6)

# probe forward
augmented = []
latencies = []
errors = 0

with torch.no_grad():
    for i, rec in enumerate(records):
        try:
            text = rec.get('input', '') + ' ' + rec.get('completion', '')
            tokens = sp.encode(text)[:T]  # truncate to T=64
            if len(tokens) < 2:
                tokens = tokens + [0] * (2 - len(tokens))
            # pad to T
            if len(tokens) < T:
                tokens = tokens + [0] * (T - len(tokens))
            idx = torch.tensor([tokens], dtype=torch.long, device=device)

            t0 = time.time()
            out = decoder(idx)
            torch.cuda.synchronize() if device.type == 'cuda' else None
            t_fwd = time.time() - t0

            # extract tensions (3rd output)
            tensions = out[2] if len(out) >= 3 else None
            if tensions is None:
                tension_target = [0.0] * T
            else:
                # tensions could be tensor [B, T] or [B, T, ...]
                if isinstance(tensions, torch.Tensor):
                    t_flat = tensions.detach().float().cpu().view(tensions.shape[0], -1)
                    # take per-position mean if multi-dim
                    if t_flat.shape[1] >= T:
                        tension_target = t_flat[0, :T].tolist()
                    else:
                        # broadcast / pad
                        v = t_flat[0].mean().item()
                        tension_target = [v] * T
                elif isinstance(tensions, list):
                    tension_target = [float(t) if not torch.is_tensor(t) else float(t.mean()) for t in tensions[:T]]
                    if len(tension_target) < T:
                        tension_target = tension_target + [0.0] * (T - len(tension_target))
                else:
                    tension_target = [0.0] * T

            latencies.append(t_fwd)
            rec_aug = dict(rec)
            rec_aug['tension_target'] = tension_target
            rec_aug['_probe_latency_s'] = t_fwd
            augmented.append(rec_aug)

            if i < 3 or i % 20 == 0:
                print(f"[probe] {i+1}/{len(records)} t_fwd={t_fwd*1000:.1f}ms tension_mean={sum(tension_target)/len(tension_target):.4f}", flush=True)
        except Exception as e:
            errors += 1
            if errors <= 3:
                print(f"[probe] err at {i}: {e}", flush=True)
            rec_aug = dict(rec)
            rec_aug['tension_target'] = None
            rec_aug['_probe_error'] = str(e)
            augmented.append(rec_aug)

# write augmented
with open(OUTPUT, 'w') as f:
    for r in augmented:
        f.write(json.dumps(r, ensure_ascii=False) + '\n')

# latency stats
if latencies:
    lat_mean = sum(latencies) / len(latencies)
    lat_p50 = sorted(latencies)[len(latencies)//2]
    lat_p95 = sorted(latencies)[int(len(latencies)*0.95)]
    proj_50k_s = lat_mean * 50000
    proj_50k_hr = proj_50k_s / 3600
else:
    lat_mean = lat_p50 = lat_p95 = proj_50k_s = proj_50k_hr = 0

stats = {
    "phase": "p9_p0_probe_clm_v4_tension",
    "n_records": len(records),
    "n_augmented": len(augmented),
    "n_errors": errors,
    "latency_ms_mean": lat_mean * 1000,
    "latency_ms_p50": lat_p50 * 1000,
    "latency_ms_p95": lat_p95 * 1000,
    "proj_50k_seconds": proj_50k_s,
    "proj_50k_hours": proj_50k_hr,
    "device": str(device),
    "ckpt_path": CLM_CKPT,
    "n_params": n_params,
    "vocab_size": vocab_size,
    "T": T,
}
with open(LATENCY_LOG, 'w') as f:
    json.dump(stats, f, indent=2)
print(f"[probe] DONE", flush=True)
print(json.dumps(stats, indent=2), flush=True)
